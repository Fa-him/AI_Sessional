#include <bits/stdc++.h> 
using namespace std; 
int main() { 
    int n, m; 
    cin >> n >> m; 
    vector<pair<int,int>> adj[100]; 
    for(int i=0,u,v,w;i<m;i++) { 
        cin >> u >> v >> w; 
        adj[u].push_back({v,w}); 
        adj[v].push_back({u,w}); 
    } 
    vector<int> h(n), g(n, 1e9), parent(n, -1); 
    for(int i=0;i<n;i++) 
        cin >> h[i]; 
    int start, goal; 
    cin >> start >> goal; 
    priority_queue<pair<int,int>, 
                   vector<pair<int,int>>, 
                   greater<pair<int,int>>> pq; 
    g[start] = 0; 
    pq.push({h[start], start}); 
    while(!pq.empty()) { 
        int u = pq.top().second; 
        pq.pop(); 
        if(u == goal) break; 
        for(auto x : adj[u]) { 
            int v = x.first; 
            int cost = x.second; 
            if(g[u] + cost < g[v]) { 
                g[v] = g[u] + cost; 
                parent[v] = u; 
                pq.push({g[v] + h[v], v}); 
            } 
        } 
    } 
    cout << "Cost = " << g[goal] << endl; 
    vector<int> path; 
    for(int v = goal; v != -1; v = parent[v]) 
        path.push_back(v); 
    reverse(path.begin(), path.end()); 
    cout << "Path: "; 
    for(int x : path) 
        cout << x << " "; 
}

#include <iostream>
using namespace std;

int n, m, numberOfColors;
int graph[101][101] = {0};
char colors[20], nodeColor[101];

bool solve(int node)
{
    if (node > n)
        return true;

    for (int c = 0; c < numberOfColors; c++)
    {
        bool canUse = true;

        for (int i = 1; i <= n; i++)
        {
            if (graph[node][i] == 1 &&
                (i == node || nodeColor[i] == colors[c]))
            {
                canUse = false;
                break;
            }
        }

        if (canUse)
        {
            nodeColor[node] = colors[c];

            if (solve(node + 1))
                return true;

            nodeColor[node] = 'N';
        }
    }

    return false;
}

int main()
{
    cin >> n >> m;

    for (int i = 0; i < m; i++)
    {
        int u, v;
        cin >> u >> v;

        graph[u][v] = 1;
        graph[v][u] = 1;
    }

    cin >> numberOfColors;

    for (int i = 0; i < numberOfColors; i++)
        cin >> colors[i];

    for (int i = 1; i <= n; i++)
        nodeColor[i] = 'N';

    if (solve(1) == false)
    {
        cout << "NO SOLUTION" << endl;
        return 0;
    }

    int usedColors = 0;

    for (int c = 0; c < numberOfColors; c++)
    {
        for (int i = 1; i <= n; i++)
        {
            if (nodeColor[i] == colors[c])
            {
                usedColors++;
                break;
            }
        }
    }

    cout << "Colors Used -> " << usedColors << endl;

    for (int i = 1; i <= n; i++)
    {
        cout << "Color of Node " << i
             << " -> " << nodeColor[i] << endl;
    }

    return 0;
}

# include <bits/stdc++.h>
using namespace std;

struct Node{
    int value;
    bool isMax;

    Node* left;
    Node* right;
};


int minimax(Node* node)
{
    if (node->left == NULL && node->right==NULL)
        return node->value;

    int leftVal= minimax(node->left);
    int rightVal= minimax(node->right);

    if(node->isMax)
        node->value=max(leftVal, rightVal);
    else
        node->value=min(leftVal, rightVal);

    return node->value;
}


int main()
{
    Node* n= NULL;

    Node L1= {-1, false, n, n};
    Node L2= {8, false, n, n};
    Node L3= {-3, false, n, n};
    Node L4= {-1, false, n, n};
    Node L5= {2, false, n, n};
    Node L6= {1, false, n, n};
    Node L7= {-3, false, n, n};
    Node L8= {4, false, n, n};


    Node D= {0, true, &L1, &L2};
    Node E= {0, true, &L3, &L4};
    Node F= {0, true, &L5, &L6};
    Node G= {0, true, &L7, &L8};

    Node B= {0, false, &D, &E};
    Node C= {0, false, &F, &G};

    Node A= {0, true, &B, &C};

    int utility= minimax(&A);

    cout<< "Utility = "<<utility<<endl;

    cout<< "Path: A-> ";

    if (A.value==B.value)
    {
        cout<< "B->";
        if (B.value==D.value)
            cout<<"D-> "<<D.value<<endl;
        else
            cout<<"E-> "<<E.value<<endl;
    }
    else
    {
        cout<<"C-> ";
        if (C.value==F.value)
            cout<<"F-> "<<F.value<<endl;
        else
            cout<<"G-> "<<G.value<<endl;
    }

    return 0;

}
